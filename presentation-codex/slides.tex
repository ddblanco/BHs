\begin{frame}{Measuring the shift of extremality}
\smallnote{Classical Einstein--Gauss--Bonnet gravity in five dimensions, $G_5=1$}
\vspace{4mm}
\[ I=\frac{1}{16\pi}\int d^5x\sqrt{-g}\,[R+\alpha\mathcal L_{\rm GB}] \]
\vspace{2mm}
\[ dM=T\,dS+2\Omega_H\,dJ+\Psi\,d\alpha\qquad\color{accent}\left.\frac{\partial M_{\rm ext}}{\partial\alpha}\right|_J=\lim_{T\to0}\Psi \]
\vspace{5mm}
\begin{columns}[T,onlytextwidth]
\begin{column}{.48\textwidth}
\key{Nonlinear solutions}\\[3mm]Four radial functions\\Chebyshev collocation + Newton\\\Nstates\ states, \Ncouplings\ couplings
\end{column}
\begin{column}{.48\textwidth}
\key{Response at each solution}\\[3mm]$R(u,\alpha)=0$\\[2mm]$R_u\,u_{,\alpha}=-R_{,\alpha}$\\[2mm]One linear solve, then extrapolate
\end{column}
\end{columns}
\vspace{7mm}
\smallnote{Standard first law and numerical methods. Project contribution: finite-coupling response.\\The extremal identity assumes a regular branch and finite $\partial_\alpha S_{\rm ext}|_J$.}
\end{frame}
